def clean():
    for p in pathlib.Path(".").rglob("__pycache__"): shutil.rmtree(p, ignore_errors=True)
def run(k):
    r=subprocess.run([sys.executable,"-m","pytest","-q","-p","no:cacheprovider","tests","-k",k],capture_output=True,text=True)
    return [l for l in r.stdout.splitlines() if l.startswith("FAILED") or " passed" in l or " failed" in l]
only=sys.argv[1:]
for name,f,old,new,k in M:
    if only and name not in only: continue
    src=open(f).read(); assert src.count(old)==1,(name,src.count(old))
    open(f,"w").write(src.replace(old,new)); clean()
    try:
        out=run(k)
    finally:
        open(f,"w").write(src); clean()
    print(f"=== {name}: {f}\n    - {old!r}\n    + {new!r}"); print("\n".join("    "+l for l in out), flush=True)
